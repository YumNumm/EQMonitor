# v2移行スキップ + アップデートバナー + Changelog Markdown化 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 移行成功ユーザーはオンボーディングをスキップし、ホームで「アップデートしました」バナー→Changelog へ誘導し、Changelog を GitHub Releases 由来の Markdown で表示する。

**Architecture:** バックエンド（Hono + Valibot）の Changelog API に Markdown `content` を追加（GitHub Releases から生成する `CHANGELOG_CONTENT` マップを datasource でマージ）。OpenAPI 再生成 → Dart クライアント再生成。アプリは Changelog を Markdown 描画し、既存 `WhatsNewBanner` をローカル版数差トリガー + Changelog 遷移に転用、通知権限バナーを新設、welcome ステップで移行成功時に即オンボーディング完了→ホーム遷移する。

**Tech Stack:** Flutter (Riverpod + Hooks + Freezed), Dart; TypeScript (Hono, Valibot, tsx, vitest); swagger_parser + build_runner。

## Global Constraints

- Flutter `3.44.0` / Dart `^3.11.0`（`mise.toml`）。
- `melos run analyze`（`dart analyze`）は警告ゼロで通ること。`dart format` 準拠（CI 強制）。
- クロスパッケージ参照は `package:` インポート（相対禁止）。
- 生成コード（`*.g.dart` / `*.freezed.dart`）は手編集しない。アノテーション変更後は `melos run generate`（または該当パッケージで `dart run build_runner build --delete-conflicting-outputs`）で再生成。
- PR ベースは `develop`、リモートは常に `origin`（`YumNumm/EQMonitor`）。
- バックエンド: Hono + Valibot、`oxlint` + `oxfmt`。スクリプトは `tsx <path>` 規約。
- `backend/api/api/openapi.json` は `generate:openapi` の正準出力で **oxfmt 対象外**（再フォーマット禁止・`JSON.stringify(specs, null, 2)` 形式を維持）。

```typescript
import type { ChangelogEntry } from '@eqmonitor-backend/changelog';
import type { ChangelogResponse } from '../model/responses';

import { createHash } from 'node:crypto';

import { CHANGELOG, CHANGELOG_CONTENT } from '@eqmonitor-backend/changelog';

function compareSemver(a: string, b: string): number {
  const pa = a.split('.').map(Number);
  const pb = b.split('.').map(Number);
  for (let i = 0; i < 3; i++) {
    const diff = (pa[i] ?? 0) - (pb[i] ?? 0);
    if (diff !== 0) return diff;
  }
  return 0;
}

export class ChangelogDatasource {
  constructor(
    private readonly changelog: ChangelogEntry[] = CHANGELOG,
    private readonly contentByVersion: Record<string, string> = CHANGELOG_CONTENT,
  ) {}

  build(options: { since?: string; limit: number }): {
    body: ChangelogResponse;
    etag: string;
  } {
    let entries = [...this.changelog];
    if (options.since) {
      const since = options.since;
      entries = entries.filter(e => compareSemver(e.version, since) > 0);
    }
    entries = entries.slice(0, options.limit);

    const body: ChangelogResponse = {
      entries: entries.map(e => ({
        version: e.version,
        date: e.date,
        url: e.url,
        sections: e.sections,
        content: this.contentByVersion[e.version],
      })),
    };
    const etag = this.computeEtag(body);
    return { body, etag };
  }

  private computeEtag(body: unknown): string {
    const json = JSON.stringify(body);
    return `"${createHash('sha256').update(json).digest('hex').slice(0, 16)}"`;
  }
}
```

- [ ] **Step 3: transform を実装**

Create `backend/packages/changelog/src/generate.ts`:

```typescript
export interface GitHubRelease {
  tag_name: string;
  body: string | null;
  draft: boolean;
  prerelease: boolean;
}

function normalizeVersion(tag: string): string | null {
  const m = tag.match(/(\d+\.\d+\.\d+)/);
  return m ? m[1] : null;
}

/**
 * GitHub Releases のリストから version -> markdown 本文 のマップを作る。
 * draft と 本文が空のリリースはスキップする。prerelease(beta/alpha) は含める。
 */
export function buildChangelogContentMap(
  releases: GitHubRelease[],
): Record<string, string> {
  const map: Record<string, string> = {};
  for (const release of releases) {
    if (release.draft) continue;
    const version = normalizeVersion(release.tag_name);
    if (!version) continue;
    const body = (release.body ?? '').trim();
    if (!body) continue;
    map[version] = body;
  }
  return map;
}
```

```typescript
import { writeFileSync } from 'node:fs';

import { buildChangelogContentMap, type GitHubRelease } from '../src/generate';

const REPO = 'YumNumm/EQMonitor';

async function fetchAllReleases(): Promise<GitHubRelease[]> {
  const headers: Record<string, string> = {
    Accept: 'application/vnd.github+json',
    'User-Agent': 'eqmonitor-changelog-generator',
  };
  const token = process.env.GITHUB_TOKEN;
  if (token) headers.Authorization = `Bearer ${token}`;

  const all: GitHubRelease[] = [];
  for (let page = 1; page <= 20; page++) {
    const res = await fetch(
      `https://api.github.com/repos/${REPO}/releases?per_page=100&page=${page}`,
      { headers },
    );
    if (!res.ok) {
      throw new Error(`GitHub API ${res.status}: ${await res.text()}`);
    }
    const batch = (await res.json()) as GitHubRelease[];
    all.push(...batch);
    if (batch.length < 100) break;
  }
  return all;
}

async function main(): Promise<void> {
  const releases = await fetchAllReleases();
  const map = buildChangelogContentMap(releases);
  const entries = Object.entries(map).sort(([a], [b]) =>
    b.localeCompare(a, undefined, { numeric: true }),
  );
  const rendered = entries
    .map(([version, content]) => `  ${JSON.stringify(version)}: ${JSON.stringify(content)},`)
    .join('\n');
  const file = `// GENERATED CODE - DO NOT EDIT BY HAND
// Generated by scripts/generate.ts from GitHub Releases (${REPO})

export const CHANGELOG_CONTENT: Record<string, string> = {
${rendered}
};
`;
  writeFileSync(new URL('../src/content.generated.ts', import.meta.url), file);
  console.log(`Wrote ${entries.length} changelog content entries`);
}

await main();
```

/// 通知権限が未許可の場合にホームシートへ表示するバナー。
/// タップで許可要求 → まだ未許可なら OS 設定を開く。閉じるボタンで dismiss。
class NotificationPermissionBanner extends ConsumerWidget {
  const NotificationPermissionBanner({required this.bottomSpacing, super.key});

  final double bottomSpacing;

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final granted =
        ref.watch(isNotificationPermissionGrantedProvider).value ?? true;
    final dismissed =
        ref.watch(notificationPermissionBannerDismissedProvider).value ?? false;
    if (granted || dismissed) {
      return const SizedBox.shrink();
    }

    final designSystem = context.designSystem;
    final colorTheme = designSystem.colorTheme;
    final spacing = designSystem.spacing;

    return Padding(
      padding: EdgeInsets.only(bottom: bottomSpacing),
      child: Material(
        color: colorTheme.primaryContainer,
        borderRadius: RoundedSuperellipseBorder(
          borderRadius: BorderRadius.circular(designSystem.shape.card),
        ).borderRadius,
        clipBehavior: Clip.antiAlias,
        child: Padding(
          padding: EdgeInsets.symmetric(
            horizontal: spacing.md,
            vertical: spacing.sm,
          ),
          child: Row(
            spacing: spacing.md,
            children: [
              Icon(
                Icons.notifications_off_rounded,
                color: colorTheme.onPrimaryContainer,
                size: 24,
              ),
              Expanded(
                child: InkWell(
                  onTap: () async {
                    final ok = await ref
                        .read(permissionRepositoryProvider)
                        .requestNotificationPermission();
                    ref.invalidate(isNotificationPermissionGrantedProvider);
                    if (!ok) {
                      await AppSettings.openAppSettings(
                        type: AppSettingsType.notification,
                      );
                    }
                  },
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      Text(
                        '通知が許可されていません',
                        style: Theme.of(context).textTheme.titleSmall?.copyWith(
                          color: colorTheme.onPrimaryContainer,
                        ),
                      ),
                      Text(
                        '緊急地震速報などの通知を受け取るには通知を許可してください',
                        style: Theme.of(context).textTheme.bodySmall?.copyWith(
                          color: colorTheme.onPrimaryContainer,
                        ),
                      ),
                    ],
                  ),
                ),
              ),
              IconButton(
                icon: Icon(
                  Icons.close,
                  color: colorTheme.onPrimaryContainer,
                  size: 20,
                ),
                tooltip: '閉じる',
                onPressed: () => unawaited(
                  ref
                      .read(
                        notificationPermissionBannerDismissedProvider.notifier,
                      )
                      .dismiss(),
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
